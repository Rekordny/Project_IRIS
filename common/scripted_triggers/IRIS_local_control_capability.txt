#当地修正等级判定 (地区判断一般用不上,且相反)
IRIS_region_modifier_level_check = {
    ROOT = {
        meta_trigger = {
            text = {
                NOT = { check_variable = { control_modifier@[tag] = token:[idea] } }
            }
            tag = "[?TAG_regional_control.GetTag]"
            idea = "[?idea.GetTokenKey]"
        }
    }
}

#当地修正等级判定 国家状态用
IRIS_region_modifier_level_check_c = {
    hidden_trigger = { ROOT = { controls_state = PREV } }
    THIS = {
        meta_trigger = {
            text = {
                custom_override_tooltip = {
                    tooltip = {
                        localization_key = IRIS_region_modifier_level_check_c_TT
                        INDENTATION = ""
                        level = [idea]
                    }
                    check_variable = { control_modifier@[tag] = token:[idea] }
                }
            }
            tag = "[?TAG_regional_control.GetTag]"
            idea = "[?idea.GetTokenKey]"
        }
    }
}

    has_variable = level_now@ROOT
