#初始化地区控制力等级
IRIS_initialize_level_regional_control = {
    if = {
        limit = { NOT = { has_variable = level_now@ROOT } }
        add_to_array = {
            array = global.IRIS_has_region_control_mechanism
            value = THIS
        }
        # IRIS_remove_region_control_mechanism = yes
        clear_variable = level_now_split@ROOT
        clear_variable = level_now@ROOT
        clear_variable = level_progress_now@ROOT
        clear_variable = IRIS_start_local_control_capability_damage_bar_frame@ROOT
        clear_variable = building_picture_click@ROOT
        set_variable = { region_bill@ROOT = token:IRIS_token_NULL }
        set_variable = { control_modifier@ROOT = token:IRIS_token_NULL }
        clear_array = scale_grid_array_x@ROOT
        clear_array = scale_grid_array@ROOT
        clear_array = policy_grid_array@ROOT
        clear_array = policy_grid_frame_array@ROOT
        set_variable = { TAG_regional_control = event_target:system }

        custom_effect_tooltip = IRIS_initialize_level_regional_control_TT
        set_temp_variable = { temp_levels = temp_levels_array^num }
        add_to_temp_variable = { temp_levels = -1 }
        clear_array = scale_grid_array@ROOT
        for_loop_effect = {
            start = 0
            end = temp_levels
            IRIS_add_segmentation_regional_control_level = yes
        }
        clear_array = regional_control_level_array@ROOT
        for_loop_effect = {
            end = temp_levels_array^num
            value = i
            add_to_array = { regional_control_level_array@ROOT = temp_levels_array^i }
        }
        add_to_array = { policy_grid_array@ROOT = token:IRIS_state_no_garrison }
        for_loop_effect = {
            end = temp_policy_grid_array^num
            value = i
            add_to_array = { policy_grid_array@ROOT = temp_policy_grid_array^i }
        }
        hidden_effect = {
            #设定法案
            if = {
                limit = { has_variable = policy_temp }
                log = "ceshi"
                set_variable = { TAG_regional_control = ROOT }
                clear_array = this.policy_grid_frame_array@ROOT
                for_loop_effect = {
                    end = this.policy_grid_array@ROOT^num
                    value = i
                    if = {
                        limit = { check_variable = { this.policy_grid_array@ROOT^i = policy_temp } }
                        set_temp_variable = { temp_i = i }
                    }
                    add_to_array = { this.policy_grid_frame_array@ROOT = 2 }
                }
                clear_variable = region_bill@ROOT

                set_variable = { this.policy_grid_frame_array@ROOT^temp_i = 1 }
                meta_effect = {
                    text = {
                        set_variable = { region_bill@[tag] = token:[region_bill] }
                        add_dynamic_modifier = { modifier = [region_bill]_modifiers }
                    }
                    region_bill = "[?policy_temp.GetTokenKey]"
                    tag = "[?ROOT.GetTag]"
                }
            }
            else_if = {
                limit = { check_variable = { policy_grid_array@ROOT^num > 0 } }
                set_variable = { TAG_regional_control = ROOT }
                set_variable = { policy_temp = policy_grid_array@ROOT^0 }
                for_loop_effect = {
                    end = this.policy_grid_array@ROOT^num
                    add_to_array = { this.policy_grid_frame_array@ROOT = 2 }
                }
                clear_variable = region_bill@ROOT

                set_variable = { this.policy_grid_frame_array@ROOT^0 = 1 }
                meta_effect = {
                    text = {
                        set_variable = { region_bill@[tag] = token:[region_bill] }
                        add_dynamic_modifier = { modifier = [region_bill]_modifiers }
                    }
                    region_bill = "[?policy_temp.GetTokenKey]"
                    tag = "[?ROOT.GetTag]"
                }
            }
        }
        clear_variable = temp_levels
        clear_variable = temp_i
        clear_variable = policy_temp
        clear_array = temp_levels_array
        clear_array = temp_policy_grid_array
    }
}
#添加地区控制力等级分段
IRIS_add_segmentation_regional_control_level = {
    add_to_array = { scale_grid_array@ROOT = 1 }
    clear_array = scale_grid_array_x@ROOT
    #x = (i + 1) * W / (n + 1)
    for_loop_effect = {
        start = 0
        end = scale_grid_array@ROOT^num
        value = temp_i

        set_temp_variable = { i = temp_i }
        add_to_temp_variable = { i = 1 }
        set_temp_variable = { x = @PROGRESS_BAR_WIDTH }
        set_temp_variable = { n = scale_grid_array@ROOT^num }
        add_to_temp_variable = { n = 1 }
        divide_temp_variable = { x = n }
        multiply_temp_variable = { x = i }
        add_to_array = { scale_grid_array_x@ROOT = x }
    }
    IRIS_region_control_initial_progress = yes
}

                            id = "[?THIS.GetID]"
                            tag = "[?ROOT.GetTag]"
                            bill = "[?bill_temp.GetTokenKey]"
                        }
                    }
                    add_to_array = { policy_grid_frame_array@ROOT = 1 }
                }
                else = {
                    add_to_array = { policy_grid_frame_array@ROOT = 2 }
                }
            }
        }
    }
}
