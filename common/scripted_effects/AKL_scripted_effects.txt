    multiply_temp_variable = {AKL_price_temp = AKL_stock_amount_selected}
    add_to_variable = {AKL_money = AKL_price_temp}

    subtract_from_temp_variable = {AKL_revenue_temp = AKL_stock_cost_selected}

        limit = {

        }

        set_temp_variable = {AKL_pp_income_temp = AKL_revenue_temp}

        multiply_temp_variable = {AKL_pp_income_temp = 0.01}
        round_temp_variable = AKL_pp_income_temp
        add_political_power = var:AKL_pp_income_temp

    }

    if = {
        limit = {
            check_variable = {AKL_goods_selected = token:AKL_weapon}
            check_variable = {AKL_location_selected = token:AKL_SAO}
        }
        SKR = {
            country_event = {id = AKL.47 days = 3}
        }
    }

    #输入记录Log
    add_to_array = {AKL_trade_log_o = AKL_location_selected}
    add_to_array = {AKL_trade_log_t = AKL_stock_time_selected}
    add_to_array = {AKL_trade_log_e = 0}#销售 支出为0
    add_to_array = {AKL_trade_log_i = AKL_price_temp}
    add_to_array = {AKL_trade_log_m = AKL_money}

    #移除销售项目
    remove_from_array = {
        array = AKL.AKL_stock_Array
        index = i
    }
    remove_from_array = {
        array = AKL.AKL_stock_Array_y
        index = i
    }
    remove_from_array = {
        array = AKL.AKL_stock_Array_p
        index = i
    }
    remove_from_array = {
        array = AKL.AKL_stock_Array_o
        index = i
    }
    remove_from_array = {
        array = AKL.AKL_stock_Array_t
        index = i
    }
    remove_from_array = {
        array = AKL.AKL_stock_Array_v
        index = i
    }

    #清空变量
    clear_variable = AKL_stock_amount_selected
    clear_variable = AKL_stock_cost_selected
    clear_variable = AKL_stock_value_selected
    clear_variable = AKL_stock_origin_selected
    clear_variable = AKL_stock_time_selected
