#替换小换大
defined_text = {
    name = GetIncreaseRegionControlModifierLevel
    text = {
        trigger = {
            NOT = { check_variable = { this.level_progress_now@ROOT > 50 } }
        }
        localization_key = "[?regional_control_level_array@ROOT^level_now@ROOT.GetTokenKey]"
    }
    text = {
        trigger = {
            set_temp_variable = { GetIncreaseRegionControlModifierLevelTemp = this.level_now@ROOT }
            add_to_temp_variable = { GetIncreaseRegionControlModifierLevelTemp = -1 }
        }
        localization_key = "[?regional_control_level_array@ROOT^GetIncreaseRegionControlModifierLevelTemp.GetTokenKey]"
    }
}
