#替换小换大
defined_text = {
    name = GetIncreaseRegionControlModifierLevel
    text = {
        trigger = {
            NOT = { check_variable = { this.level_progress_now > 50 } }
        }
        localization_key = "[?regional_control_level_array^level_now.GetTokenKey]"
    }
    text = {
        trigger = {
            set_temp_variable = { GetIncreaseRegionControlModifierLevelTemp = this.level_now }
            add_to_temp_variable = { GetIncreaseRegionControlModifierLevelTemp = -1 }
        }
        localization_key = "[?regional_control_level_array^GetIncreaseRegionControlModifierLevelTemp.GetTokenKey]"
    }
}
