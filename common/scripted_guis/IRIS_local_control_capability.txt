scripted_gui = {
    start_local_control_capability = {
        context_type = selected_state_context
        parent_window_token = selected_state_view
        window_name = "start_local_control_capability"
        visible = {
            # always = no
            check_variable = { this.scale_grid_array@ROOT^num > 0 }
            is_in_array = { global.IRIS_has_region_control_mechanism = THIS }
        }
        properties = {
            scale = {
                x = this.scale_grid_array_x@ROOT^scale_grid_array_index
                # frame = OCF_gridbox_seat_array_senate^idx
            }
            damage_bar = {
                image = "[IRIS_start_local_control_capability_damage_bar_image]"
                frame = temp_var:this.IRIS_start_local_control_capability_damage_bar_frame@ROOT
            }
        }
        triggers = {
            arrow_pointer_animation_visible = { NOT = { check_variable = { building_picture_click > 0 } } }
            damage_bar_visible = {
                set_temp_variable = { IRIS_start_local_control_capability_damage_bar_frame@ROOT = level_now@ROOT }
                multiply_temp_variable = { IRIS_start_local_control_capability_damage_bar_frame@ROOT = 100 }
                divide_temp_variable = { IRIS_start_local_control_capability_damage_bar_frame@ROOT = regional_control_level_array@ROOT^num }
            }
        }
        effects = {

            building_picture_click = {
                if = {
                    limit = {
                        check_variable = { ROOT.local_control_capability_var = 1 }
                    }
                    set_variable = { ROOT.local_control_capability_var = 0 }
                }
                else = {
                    set_variable = { ROOT.local_control_capability_var = 1 }
                }
                add_to_variable = { building_picture_click = 1 }
            }
        }
    }
    local_control_capability = {
        context_type = selected_state_context
        parent_window_token = selected_state_view
        window_name = "local_control_capability"
        visible = {
            check_variable = { ROOT.local_control_capability_var = 1 }
            check_variable = { this.scale_grid_array@ROOT^num > 0 }
            is_in_array = { global.IRIS_has_region_control_mechanism = THIS }
        }
        dynamic_lists = {
            # 进度刻度
            scale_grid = {
                array = this.scale_grid_array@ROOT
                index = scale_grid_array_index
                value = scale_grid_array_value
                entry_container = scale_slot
                # change_scope = no
            }
            # 政策
            policy_grid = {
                array = this.policy_grid_array@ROOT
                index = policy_grid_array_index
                value = policy_grid_array_value
                entry_container = policy_slot
                # change_scope = no
            }
        }
        properties = {
            scale = {
                x = this.scale_grid_array_x@ROOT^scale_grid_array_index
                # frame = OCF_gridbox_seat_array_senate^idx
            }
            policy = { image = "GFX_[?policy_grid_array_value.GetTokenKey]" }
            policy_bg = { frame = this.policy_grid_frame_array@ROOT^policy_grid_array_index }
            region_occupation_progress = { frame = this.level_progress_now@ROOT }

            left_icon = { image = "GFX_[?THIS.regional_control_level_array@ROOT^left_icon_temp.GetTokenKey]" }
            right_icon = { image = "GFX_[?THIS.regional_control_level_array@ROOT^THIS.level_now@ROOT.GetTokenKey]" }
        }
        triggers = {
            bg_coverage_visible = {
                NOT = {
                    ROOT = {
                        controls_state = PREV
                        meta_trigger = {
                            text = {
                                [bill]_trigger = yes
                                set_temp_variable = { bill_pp = global.[bill]_pp }
                                if = {
                                    limit = { check_variable = { bill_pp = 0 } }
                                    set_temp_variable = { bill_pp = 15 }
                                }
                                hidden_trigger = { NOT = { has_political_power < temp_var:bill_pp } }
                            }
                            bill = "[?policy_grid_array_value.GetTokenKey]"
                        }
                    }
                }
            }
            bg_coverage_enabled = {
                ROOT = {
                    controls_state = PREV
                    meta_trigger = {
                        text = {
                            [trigger]_trigger = yes
                            set_temp_variable = { bill_pp = global.[bill]_pp }
                            if = {
                                limit = { check_variable = { bill_pp = 0 } }
                                set_temp_variable = { bill_pp = 15 }
                            }
                            # hidden_trigger = { NOT = { has_political_power < temp_var:bill_pp } }
                        }
                        trigger = "[?policy_grid_array_value.GetTokenKey]"
                        bill = "[?policy_grid_array_value.GetTokenKey]"
                    }
                }
            }

            left_icon_glow_visible = { NOT = { check_variable = { this.level_progress_now@ROOT > 50 } } }
            right_icon_glow_visible = { check_variable = { this.level_progress_now@ROOT > 50 } }
            left_icon_visible = {
                set_temp_variable = { left_icon_temp = THIS.level_now@ROOT }
                add_to_temp_variable = { left_icon_temp = -1 }
            }
        }
        effects = {

            level_tt_click = {
                # set_temp_variable = { this.level_tt_click_temp = this.level_now }
                meta_effect = {
                    text = {
                        ROOT = { add_ideas = [modifier] }
                    }
                    modifier = "[?this.control_modifier@ROOT.GetTokenKey]"
                }
            }
            left_icon_click = {
                if = {
                    limit = { check_variable = { this.level_progress_now@ROOT > 50 } }
                    meta_effect = {
                        text = {
                            ROOT = {
                                swap_ideas = {
                                    remove_idea = [NowLevel]
                                    add_idea = [NextLevel]
                                }
                            }
                        }
                        NextLevel = "[GetIncreaseRegionControlModifierLevel]"
                        NowLevel = "[GetLocalControlCapabilityScaleNowLevel]"
                    }
                }
                else = {
                    meta_effect = {
                        text = {
                            ROOT = {
                                swap_ideas = {
                                    remove_idea = [NowLevel]
                                    add_idea = [NowLevel]
                                }
                                custom_effect_tooltip = IRIS_local_control_capability_level_pdx_tooltip_null
                            }
                        }
                        NowLevel = "[?this.control_modifier@ROOT.GetTokenKey]"
                    }
                }
            }
            right_icon_click = {
                if = {
                    limit = { NOT = { check_variable = { this.level_progress_now@ROOT > 50 } } }
                    meta_effect = {
                        text = {
                            ROOT = {
                                swap_ideas = {
                                    remove_idea = [NextLevel]
                                    add_idea = [NowLevel]
                                }
                            }
                        }
                        NextLevel = "[GetDecreaseRegionControlModifierLevel]"
                        NowLevel = "[GetLocalControlCapabilityScaleNowLevel]"
                    }
                }
                else = {
                    meta_effect = {
                        text = {
                            ROOT = {
                                swap_ideas = {
                                    remove_idea = [NowLevel]
                                    add_idea = [NowLevel]
                                }
                                custom_effect_tooltip = IRIS_local_control_capability_level_pdx_tooltip_null
                            }
                        }
                        NowLevel = "[?this.control_modifier@ROOT.GetTokenKey]"
                    }
                }
            }
            policy_bg_click = {
                if = {
                    limit = { NOT = { check_variable = { this.policy_grid_frame_array@ROOT^policy_grid_array_index = 1 } } }
                    hidden_effect = { ROOT = { add_political_power = -15 } }
                    IRIS_refresh_region_control_bill = yes
                }
            }
        }
    }
